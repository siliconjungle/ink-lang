use ink_compiled_representation::State;
use serde_json::json;
use std::{hint::black_box,time::Instant};
fn main(){let boolean=std::env::args().nth(1).as_deref()==Some("boolean");for n in [16u64,128,1024]{
 let mut state=State::new().unwrap();for i in 0..n{state.invoke_json(if boolean{"insert"}else{"put"},&if boolean{json!([i,true,false])}else{json!([i,i*13])}).unwrap();}state.acknowledge_through(u64::MAX,u64::MAX);
 let phases=if boolean{vec!["any_query","all_query","insert","replace","erase"]}else{vec!["get-hot","get-cold","put","erase"]};
 for phase in phases{let iterations=match phase{"insert"=>16,"put"|"replace"|"erase"=>32,_=>512};let mut times=vec![];
 let args=|i:u64|if boolean{match phase{"insert"=>json!([n+i,true,false]),"replace"=>json!([n-1,i%2==0,false]),"erase"=>json!([n+100,false]),_=>json!([])}}else{match phase{"get-hot"=>json!([1023]),"get-cold"=>json!([n-1]),"put"=>json!([n-1,i+1]),_=>json!([n+100,false])}};
 let name=if phase.starts_with("get-"){"get"}else{phase};
 if phase!="insert"{for i in 0..16{black_box(state.invoke_json(name,&args(i)).unwrap());}}state.acknowledge_through(u64::MAX,u64::MAX);
 for sample in 0..7{let start=Instant::now();for i in 0..iterations{black_box(state.invoke_json(name,&args(i+sample*iterations)).unwrap());}times.push(start.elapsed().as_nanos() as u64/iterations);state.acknowledge_through(u64::MAX,u64::MAX);}
 println!("{}",json!({"rows":n,"phase":format!("action:{phase}"),"samples_ns":times,"iterations":iterations,"transport":"public generated Rust invoke_json, including argument/result conversion and events"}));
 }
}}
