use serde_json::{json,Value};use std::{hint::black_box,time::Instant};
use verified_language::{core::CheckedModule,representation::{self,Request},stateful::Runtime,syntax::parse};use ink_runtime::representation::{Session,Reference};
fn load(module:&CheckedModule,label:&str)->representation::CheckedRepresentation{let r:Request=serde_json::from_slice(&std::fs::read(format!("{DIRECTORY}/{label}.json")).unwrap()).unwrap();representation::check(module,&r).unwrap()}
fn call(i:usize,reads:usize,count:usize,query:&str)->(&str,Value){if i%100<reads{(query,json!([]))}else{let key=(i*17%count.max(1))as u64;match i%4{0=>("erase",json!([key,false])),1=>("put",json!([key,u64::MAX-i as u64,true,false])),2=>("put",json!([key,i,false,false])),_=>("put",json!([count as u64+i as u64,i,true,false]))}}}
fn main(){let module=CheckedModule::from_source(parse(SOURCE).unwrap()).unwrap();let labels=["direct","refresh","sum-delta","count-delta"];let plans=labels.iter().map(|s|load(&module,s)).collect::<Vec<_>>();let mut phases=vec![];let mut bridges=vec![];let layout=ink_core::snapshot::layout(module.program()).unwrap();
 let script=(0..28u64).flat_map(|i|[("put",json!([i%4,if i%3==0{u64::MAX}else{i},i%2==0,i%7==0])),("erase",json!([(i+1)%4,i%5==0])),("total_query",json!([])),("count_query",json!([]))]).collect::<Vec<_>>();
 for plan in &plans{let mut native=Session::new(plan.clone(),Native).unwrap();let mut reference=Session::new(plan.clone(),Reference).unwrap();let mut baseline=Runtime::new(module.program().clone()).unwrap();for(a,args)in &script{let expected=baseline.invoke_json(a,args).unwrap().json();assert_eq!(native.invoke_json(a,args).unwrap(),expected);assert_eq!(reference.invoke_json(a,args).unwrap(),expected);assert_eq!(native.checkpoint().unwrap(),baseline.checkpoint_portable().unwrap());assert_eq!(reference.checkpoint().unwrap(),native.checkpoint().unwrap());}let bytes=native.checkpoint().unwrap();let mut migrated=Session::restore(plan.clone(),&bytes,Native).unwrap();migrated.migrate(plans[0].clone()).unwrap();assert_eq!(migrated.checkpoint().unwrap(),bytes);let mut logical=ink_core::snapshot_wire::decode(&layout,&bytes,Default::default()).unwrap();logical.version=u64::MAX;let full=ink_core::snapshot_wire::encode(&layout,&logical,Default::default()).unwrap();let mut exhausted=Session::restore(plan.clone(),&full,Native).unwrap();assert!(exhausted.invoke_json("put",&json!([0,1,true,false])).is_err());assert_eq!(exhausted.checkpoint().unwrap(),full);}
 for(index,label)in [(2,"sum"),(3,"count")]{let bridge:Value=serde_json::from_slice(&std::fs::read(format!("{DIRECTORY}/{label}-forward.json")).unwrap()).unwrap();let bridge=representation::check_migration(&plans[0],&plans[index],&serde_json::from_value(bridge["knowledge"].clone()).unwrap(),bridge["root"].as_str().unwrap()).unwrap();let mut s=Session::new(plans[0].clone(),Native).unwrap();s.invoke_json("put",&json!([7,u64::MAX,true,false])).unwrap();let bytes=s.checkpoint().unwrap();s.migrate_direct(plans[index].clone(),&bridge).unwrap();assert_eq!(s.checkpoint().unwrap(),bytes);assert_eq!(s.invoke_json("total_query",&json!([])).unwrap()["result"],json!(u64::MAX));assert_eq!(s.invoke_json("count_query",&json!([])).unwrap()["result"],json!(1));bridges.push((index,bridge));}
 if !TIMED{return}let sizes:Vec<usize>=std::env::var("INK_BENCH_ROWS").unwrap_or("48,256,1024,2048".into()).split(',').map(|x|x.parse().unwrap()).collect();let mut measurements=vec![];
 for count in sizes{let rows=(0..count).map(|i|(json!(i),json!({"value":i as u64*13+7,"active":i%3!=0}))).collect::<Vec<_>>();let bytes=ink_core::snapshot_wire::encode(&layout,&ink_core::snapshot_wire::Logical{version:count as u64,tables:vec![rows.clone()],outbox:vec![]},Default::default()).unwrap();
  let mut order=(0..labels.len()).collect::<Vec<_>>();if std::env::var_os("INK_BENCH_REVERSE").is_some(){order.reverse();}
  for index in order{let label=labels[index];let start=Instant::now();let _=load(&module,label);let check_ns=start.elapsed().as_nanos();let start=Instant::now();let mut init=Session::restore(plans[index].clone(),&bytes,Native).unwrap();let initialise_ns=start.elapsed().as_nanos();let start=Instant::now();init.migrate(plans[(index+1)%4].clone()).unwrap();let migration_ns=start.elapsed().as_nanos();assert_eq!(init.checkpoint().unwrap(),bytes);let direct_migration_ns=bridges.iter().find(|(i,_)|*i==index).map(|(_,bridge)|{let mut from=Session::restore(plans[0].clone(),&bytes,Native).unwrap();let start=Instant::now();from.migrate_direct(plans[index].clone(),bridge).unwrap();let ns=start.elapsed().as_nanos();assert_eq!(from.checkpoint().unwrap(),bytes);ns});phases.push(json!({"rows":count,"candidate":label,"check_ns":check_ns,"initialise_ns":initialise_ns,"generic_migration_ns":migration_ns,"direct_migration_ns":direct_migration_ns}));
   for query in ["total_query","count_query"]{for reads in [100usize,90,50,10,0]{let mut samples=vec![];for round in 0..5{let mut s=Session::restore(plans[index].clone(),&bytes,Native).unwrap();let mut baseline=Runtime::restore_portable(module.program().clone(),&bytes).unwrap();let mut hand:std::collections::BTreeMap<u64,(u64,bool)>=rows.iter().map(|(k,v)|(k.as_u64().unwrap(),(v["value"].as_u64().unwrap(),v["active"].as_bool().unwrap()))).collect();let mut version=count as u64;let script=(0..100).map(|i|call(i+round*100,reads,count,query)).collect::<Vec<_>>();
    // The independent map/scanner has no cached arithmetic or Ink definitions.
    let start=Instant::now();let native=script.iter().map(|(a,args)|{let v=s.invoke_json(a,args).unwrap();s.acknowledge_through(u64::MAX,u64::MAX);black_box(v)}).collect::<Vec<_>>();let native_ns=start.elapsed().as_nanos();
    let start=Instant::now();let source=script.iter().map(|(a,args)|{let v=baseline.invoke_json(a,args).unwrap().json();baseline.acknowledge_through(u64::MAX,u64::MAX);black_box(v)}).collect::<Vec<_>>();let baseline_ns=start.elapsed().as_nanos();assert_eq!(native,source);

    let mut incremental:std::collections::BTreeMap<u64,(u64,bool)>=rows.iter().map(|(k,v)|(k.as_u64().unwrap(),(v["value"].as_u64().unwrap(),v["active"].as_bool().unwrap()))).collect();let mut sum=incremental.values().filter(|(_,active)|*active).fold(0u64,|n,(v,_)|n.wrapping_add(*v));let mut active_count=incremental.values().filter(|(_,active)|*active).count()as u64;let mut incremental_version=count as u64;
    let start=Instant::now();let maintained=script.iter().map(|(a,args)|{let read=*a==query;let result=if read{json!(if query=="total_query"{sum}else{active_count})}else{let key=args[0].as_u64().unwrap();let old=if *a=="erase"{incremental.remove(&key)}else{let row=(args[1].as_u64().unwrap(),args[2].as_bool().unwrap());let old=incremental.insert(key,row);if row.1{sum=sum.wrapping_add(row.0);active_count+=1;}old};if let Some((value,true))=old{sum=sum.wrapping_sub(value);active_count-=1;}incremental_version+=1;json!({"Ok":null})};let events=if read{json!([])}else{json!([{"commit":incremental_version,"position":0,"channel":"seen","value":if *a=="erase"{args[0].clone()}else{args[1].clone()}}])};black_box(json!({"result":result,"committed":!read,"version":incremental_version,"events":events}))}).collect::<Vec<_>>();let handwritten_incremental_ns=start.elapsed().as_nanos();assert_eq!(native,maintained);assert_eq!(s.checkpoint().unwrap(),baseline.checkpoint_portable().unwrap());samples.push(json!({"round":round,"native_ns":native_ns,"baseline_ink_ns":baseline_ns,"handwritten_rust_ns":handwritten_ns,"handwritten_incremental_rust_ns":handwritten_incremental_ns}));}measurements.push(json!({"rows":count,"candidate":label,"query":query,"read_percent":reads,"calls_per_sample":100,"samples":samples}));}}
  }
 }
 println!("{}",json!({"schema":1,"source":module.identity().unwrap(),"profile":if cfg!(debug_assertions){"debug"}else{"release"},"scope":"100 complete validated calls; insert, replace, removal, filter changes; acknowledgement per call; baseline Ink and independent handwritten BTreeMap scanner and maintained sum/count; handwritten excludes generic ingress/snapshot validation; checking/init/migration outside steady-state; native Session transport, not standalone", "phases":phases,"measurements":measurements}));
