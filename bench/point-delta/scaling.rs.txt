use serde_json::{json,Value};
use std::{alloc::{GlobalAlloc,Layout,System},hint::black_box,sync::atomic::{AtomicUsize,Ordering},time::Instant};
use verified_language::{core::CheckedModule,representation::{self,CheckedRepresentation,Request},stateful::Runtime,syntax::parse};
use ink_runtime::representation::{Executor,Reference,Session};
struct Allocator;
static CALLS:AtomicUsize=AtomicUsize::new(0);static BYTES:AtomicUsize=AtomicUsize::new(0);
static LIVE:AtomicUsize=AtomicUsize::new(0);static PEAK:AtomicUsize=AtomicUsize::new(0);
fn added(n:usize){CALLS.fetch_add(1,Ordering::Relaxed);BYTES.fetch_add(n,Ordering::Relaxed);let live=LIVE.fetch_add(n,Ordering::Relaxed)+n;PEAK.fetch_max(live,Ordering::Relaxed);}
unsafe impl GlobalAlloc for Allocator{
 unsafe fn alloc(&self,l:Layout)->*mut u8{let p=System.alloc(l);if !p.is_null(){added(l.size());}p}
 unsafe fn alloc_zeroed(&self,l:Layout)->*mut u8{let p=System.alloc_zeroed(l);if !p.is_null(){added(l.size());}p}
 unsafe fn dealloc(&self,p:*mut u8,l:Layout){System.dealloc(p,l);LIVE.fetch_sub(l.size(),Ordering::Relaxed);}
 unsafe fn realloc(&self,p:*mut u8,l:Layout,n:usize)->*mut u8{let q=System.realloc(p,l,n);if !q.is_null(){if n>=l.size(){added(n-l.size());}else{LIVE.fetch_sub(l.size()-n,Ordering::Relaxed);}}q}
}
#[global_allocator]static ALLOC:Allocator=Allocator;
struct Unfused;
impl Executor for Unfused{fn execute(&self,d:&ink_core::definition::CheckedDefinition,c:&ink_core::logic::Context,f:&str,a:Vec<ink_core::logic::Term>)->ink_core::LangResult<ink_core::logic::Term>{Native.execute(d,c,f,a)}}
fn parity<E:Executor>(plan:CheckedRepresentation,module:&CheckedModule,executor:E){
 let mut session=Session::new(plan.clone(),executor).unwrap();let mut baseline=Runtime::new(module.program().clone()).unwrap();
 for i in 0..120u64{let(action,args)=match i%12{
 0=>("put",json!([0,i])),1=>("zero",json!([])),2=>("replace_zero",json!([i+91,false])),3=>("zero",json!([])),
 4=>("replace_zero",json!([i,true])),5=>("erase_zero",json!([true])),6=>("get",json!([0])),7=>("put",json!([i+1,i])),
 8=>("erase_zero",json!([false])),9=>("zero",json!([])),10=>("replace_zero",json!([i,false])),_=>("get",json!([i]))};
 assert_eq!(session.invoke_json(action,&args).unwrap(),baseline.invoke_json(action,&args).unwrap().json());assert_eq!(session.checkpoint().unwrap(),baseline.checkpoint_portable().unwrap());
 }
 // Existing pending events, restoration and exhaustion survive the new update path.
 let bytes=session.checkpoint().unwrap();let mut restored=Session::restore(plan.clone(),&bytes,Native).unwrap();assert_eq!(restored.checkpoint().unwrap(),bytes);
 let layout=ink_core::snapshot::layout(module.program()).unwrap();let mut logical=ink_core::snapshot_wire::decode(&layout,&bytes,Default::default()).unwrap();logical.version=u64::MAX;
 let full=ink_core::snapshot_wire::encode(&layout,&logical,Default::default()).unwrap();restored=Session::restore(plan,&full,Native).unwrap();assert!(restored.invoke_json("erase_zero",&json!([false])).is_err());assert_eq!(restored.checkpoint().unwrap(),full);
}
fn timed<F:FnOnce()->T,T>(f:F)->(T,u64,usize){let initial=LIVE.load(Ordering::Relaxed);PEAK.store(initial,Ordering::Relaxed);let start=Instant::now();let t=f();let ns=start.elapsed().as_nanos() as u64;(t,ns,PEAK.load(Ordering::Relaxed).saturating_sub(initial))}
fn sample<E:Executor>(plan:CheckedRepresentation,bytes:&[u8],executor:E,action:&str,args:&Value)->(u64,usize){
 let mut session=Session::restore(plan,bytes,executor).unwrap();
 let (_,ns,peak)=timed(||{black_box(session.invoke_json(action,black_box(args)).unwrap());session.acknowledge_through(u64::MAX,u64::MAX);});(ns,peak)
}

fn main(){
 let module=CheckedModule::from_source(parse(SOURCE).unwrap()).unwrap();
 let load=|label:&str|{let request:Request=serde_json::from_slice(&std::fs::read(format!("{DIRECTORY}/{label}.json")).unwrap()).unwrap();representation::check(&module,&request).unwrap()};
 let plans=[load("direct"),load("refresh"),load("delta")];
 for p in &plans{parity(p.clone(),&module,Native);}
 let mut reports=vec![];
 let sizes:Vec<usize>=std::env::var("INK_BENCH_ROWS").unwrap_or("48,256,1024,2048".into()).split(',').map(|x|x.parse().unwrap()).collect();
 for count in sizes {
  let layout=ink_core::snapshot::layout(module.program()).unwrap();
  let logical=ink_core::snapshot_wire::Logical{version:count as u64,tables:vec![(0..count).map(|i|(json!(i),json!(i as u64*13+7))).collect()],outbox:vec![]};
  let bytes=ink_core::snapshot_wire::encode(&layout,&logical,Default::default()).unwrap();
  let mut order=vec![(0,"direct"),(1,"refresh"),(2,"delta")];if std::env::var_os("INK_BENCH_REVERSE").is_some(){order.reverse();}
  for (index,label) in order{
   let plan=&plans[index];
   let(_,check_ns,_)=timed(||black_box(load(label)));
   let(session,initialise_ns,_) = timed(||Session::restore(plan.clone(),&bytes,Native).unwrap());
   let mut migrate=session;let(_,migration_ns,_)=timed(||migrate.migrate(plans[(index+1)%3].clone()).unwrap());assert_eq!(migrate.checkpoint().unwrap(),bytes);
   drop(migrate);
   for reads in [90usize,10]{
    let mut session=Session::restore(plan.clone(),&bytes,Native).unwrap();
    let mut baseline=Runtime::restore_portable(module.program().clone(),&bytes).unwrap();
    let mut handwritten:std::collections::BTreeMap<u64,u64>=(0..count as u64).map(|i|(i,i*13+7)).collect();
    let mut version=count as u64;
    // Validate the independent handwritten algorithm and source baseline before timing.
    for i in 0..100{let read=i%100<reads;let key=(i%count.max(1)) as u64;let args=if read{json!([key])}else{json!([key,i as u64])};let action=if read{"get"}else{"put"};
     let expected=baseline.invoke_json(action,&args).unwrap().json();assert_eq!(session.invoke_json(action,&args).unwrap(),expected);
     let result=if read{handwritten.get(&key).map(|x|json!({"Some":x})).unwrap_or(json!({"None":null}))}else{handwritten.insert(key,i as u64);version+=1;json!({"Ok":null})};
     assert_eq!(json!({"result":result,"committed":!read,"version":version,"events":[]}),expected);
    }
    assert_eq!(session.checkpoint().unwrap(),baseline.checkpoint_portable().unwrap());
    let mut stages=std::collections::BTreeMap::<String,(u128,usize,usize)>::new();
    let mut stage_name="start";let mut start=Instant::now();let mut calls=CALLS.load(Ordering::Relaxed);let mut allocated=BYTES.load(Ordering::Relaxed);
    let mut observe=|name:&'static str|{let now=Instant::now();let c=CALLS.load(Ordering::Relaxed);let b=BYTES.load(Ordering::Relaxed);if stage_name!="start"&&stage_name!="done"{let e=stages.entry(stage_name.to_string()).or_default();e.0+=now.duration_since(start).as_nanos();e.1+=c-calls;e.2+=b-allocated;}stage_name=name;start=Instant::now();calls=CALLS.load(Ordering::Relaxed);allocated=BYTES.load(Ordering::Relaxed);};
    for i in 0..100{let read=i<reads;let key=(i%count.max(1))as u64;session.invoke_json_profiled(if read{"get"}else{"put"},&if read{json!([key])}else{json!([key,i])},&mut observe).unwrap();}
    let mut native_samples=vec![];let mut baseline_samples=vec![];let mut handwritten_samples=vec![];
    for round in 0..7{
     let(_,ns,peak)=timed(||{for i in 0..100{let read=i<reads;let key=(i%count.max(1))as u64;black_box(session.invoke_json(if read{"get"}else{"put"},&if read{json!([key])}else{json!([key,round+i])}).unwrap());}});native_samples.push(json!({"ns":ns,"peak_extra_bytes":peak}));
     let(_,ns,peak)=timed(||{for i in 0..100{let read=i<reads;let key=(i%count.max(1))as u64;black_box(baseline.invoke_json(if read{"get"}else{"put"},&if read{json!([key])}else{json!([key,round+i])}).unwrap().json());}});baseline_samples.push(json!({"ns":ns,"peak_extra_bytes":peak}));
     let(_,ns,peak)=timed(||{for i in 0..100{let read=i<reads;let key=(i%count.max(1))as u64;let result=if read{handwritten.get(&key).map(|x|json!({"Some":x})).unwrap_or(json!({"None":null}))}else{handwritten.insert(key,(round+i)as u64);version+=1;json!({"Ok":null})};black_box(json!({"result":result,"committed":!read,"version":version,"events":[]}));}});handwritten_samples.push(json!({"ns":ns,"peak_extra_bytes":peak}));
    }
    reports.push(json!({"rows":count,"candidate":label,"read_percent":reads,"calls_per_sample":100,"check_ns":check_ns,"initialise_ns":initialise_ns,"migration_ns":migration_ns,"stages":stages,"native":native_samples,"baseline_ink":baseline_samples,"handwritten_rust":handwritten_samples}));
   }
  }
 }
 println!("{}",json!({"schema":2,"source":module.identity().unwrap(),"profile":if cfg!(debug_assertions){"debug"}else{"release"},"scope":"steady-state complete Session calls vs baseline Ink calls; handwritten BTreeMap and complete reply envelope excludes generic input transport/snapshot validation; allocation-instrumented medians; stage profiles 100 calls; checking/initialisation/migration separated; compilation external", "measurements":reports}));
}
