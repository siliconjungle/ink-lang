use std::{cell::Cell,rc::Rc};
use verified_language::{core::CheckedModule,definition::CheckedDefinition,logic::{Context,Term},representation,stateful::Runtime,syntax::parse};
use ink_runtime::representation::{Executor,Session};
use serde_json::{json,Value};
struct Counting {encode:String,calls:Rc<Cell<usize>>}
impl Executor for Counting {
    fn execute(&self,d:&CheckedDefinition,c:&Context,f:&str,a:Vec<Term>)->ink_core::LangResult<Term>{
        if f==self.encode {self.calls.set(self.calls.get()+1);}
        Native.execute(d,c,f,a)
    }
}
struct Failing {function:String}
impl Executor for Failing {
    fn execute(&self,d:&CheckedDefinition,c:&Context,f:&str,a:Vec<Term>)->ink_core::LangResult<Term>{
        if f==self.function{return Err("injected bridge failure".into());}
        Native.execute(d,c,f,a)
    }
}
fn main(){
    let module=CheckedModule::from_source(parse(SOURCE).unwrap()).unwrap();
    let load=|label:&str|{let request=serde_json::from_slice(&std::fs::read(format!("{DIRECTORY}/{label}.json")).unwrap()).unwrap();representation::check(&module,&request).unwrap()};
    let old=load("previous");let new=load("cached");
    let bridge=|label:&str,a:&representation::CheckedRepresentation,b:&representation::CheckedRepresentation|{
        let p:Value=serde_json::from_slice(&std::fs::read(format!("{DIRECTORY}/{label}.json")).unwrap()).unwrap();
        representation::check_migration(a,b,&serde_json::from_value(p["knowledge"].clone()).unwrap(),p["root"].as_str().unwrap()).unwrap()
    };
    let forward=bridge("forward",&old,&new);let backward=bridge("backward",&new,&old);
    let calls=Rc::new(Cell::new(0));
    let mut session=Session::new(old.clone(),Counting{encode:new.plan().encode.clone(),calls:calls.clone()}).unwrap();
    let mut baseline=Runtime::new(module.program().clone()).unwrap();
    // Pending events, cached Some/None, aborts and unrelated writes survive each
    // direction of switching. The generic three-call transport is exercised too.
    for i in 0..60u64 {
        let (action,args)=match i%10 {
            0=>("put",json!([0,i])),1=>("zero",json!([])),
            2=>("flag",json!([i%3,i%2==0])),3=>("flagged",json!([i%3])),
            4=>("erase",json!([0,true])),5=>("zero",json!([])),
            6=>("put",json!([i+1,i])),7=>("get",json!([i+1])),
            8=>("erase",json!([0,false])),_=>("zero",json!([])),
        };
        assert_eq!(session.invoke_json(action,&args).unwrap(),baseline.invoke_json(action,&args).unwrap().json());
        let snapshot=baseline.checkpoint_portable().unwrap();
        calls.set(0);
        let (target,migration)=if i%2==0{(new.clone(),&forward)}else{(old.clone(),&backward)};
        session.migrate_direct(target,migration).unwrap();
        assert_eq!(calls.get(),0,"state-reusing bridge must not execute an encoder");
        assert_eq!(session.checkpoint().unwrap(),snapshot);
        let previous=session.representation().to_owned();
        assert!(session.migrate_direct(old.clone(),&forward).is_err());
        assert_eq!(session.representation(),previous);assert_eq!(session.checkpoint().unwrap(),snapshot);
    }
    calls.set(0);session.migrate(new.clone()).unwrap();assert_eq!(calls.get(),1);
    assert_eq!(session.checkpoint().unwrap(),baseline.checkpoint_portable().unwrap());
    let mut failing=Session::restore(old.clone(),&session.checkpoint().unwrap(),Failing{function:forward.function().into()}).unwrap();
    let previous=failing.checkpoint().unwrap();
    assert!(failing.migrate_direct(new.clone(),&forward).is_err());
    assert_eq!(failing.representation(),old.request().representation);assert_eq!(failing.checkpoint().unwrap(),previous);
    // Exhausted versions, including their pending outbox, also migrate exactly.
    let layout=ink_core::snapshot::layout(module.program()).unwrap();
    let mut logical=ink_core::snapshot_wire::decode(&layout,&previous,Default::default()).unwrap();logical.version=u64::MAX;
    let exhausted=ink_core::snapshot_wire::encode(&layout,&logical,Default::default()).unwrap();
    let mut session=Session::restore(old.clone(),&exhausted,Native).unwrap();
    session.migrate_direct(new.clone(),&forward).unwrap();assert_eq!(session.checkpoint().unwrap(),exhausted);
    assert!(session.invoke_json("put",&json!([0,77])).is_err());assert_eq!(session.checkpoint().unwrap(),exhausted);
    session.migrate_direct(old,&backward).unwrap();assert_eq!(session.checkpoint().unwrap(),exhausted);
}
