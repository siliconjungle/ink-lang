use std::{hint::black_box, time::Instant};
use verified_language::{core::CheckedModule, definition::CheckedDefinition, logic::{Context,Term}, representation::{self,CheckedRepresentation}, stateful::Runtime, syntax::parse};
use ink_runtime::representation::{Executor,Session};
use serde_json::{json,Value};

// The old three-call ABI, running the same native bodies in the same process.
struct Unfused;
impl Executor for Unfused {
    fn execute(&self,d:&CheckedDefinition,c:&Context,f:&str,a:Vec<Term>)->ink_core::LangResult<Term>{Native.execute(d,c,f,a)}
}
fn measure<E:Executor>(checked:CheckedRepresentation,bytes:&[u8],executor:E,action:&str,args:&Value)->Value{
    let mut session=Session::restore(checked,bytes,executor).unwrap();
    for _ in 0..32 {black_box(session.invoke_json(action,args).unwrap());session.acknowledge_through(u64::MAX,u64::MAX);}
    let mut samples=vec![];
    for _ in 0..9 {
        let start=Instant::now();
        for _ in 0..128 {black_box(session.invoke_json(action,black_box(args)).unwrap());session.acknowledge_through(u64::MAX,u64::MAX);}
        samples.push(start.elapsed().as_nanos() as u64/128);
    }
    let mut sorted=samples.clone();sorted.sort();
    json!({"samples_ns":samples,"median_ns":sorted[4],"snapshot_bytes":session.checkpoint().unwrap().len()})
}
fn main(){
    let module=CheckedModule::from_source(parse(SOURCE).unwrap()).unwrap();
    let load=|label:&str|{
        let p:Value=serde_json::from_slice(&std::fs::read(format!("{DIRECTORY}/{label}.json")).unwrap()).unwrap();
        representation::check(&module,&serde_json::from_value(p["requests"][1].clone()).unwrap()).unwrap()
    };
    let old=load("rebuild");let new=load("retain");let mut rows=vec![];
    for count in [8,24,48]{
        let mut baseline=Runtime::new(module.program().clone()).unwrap();
        // Cached key 0 is absent, so the comparison candidate refresh scans all rows.
        for key in 1..=count {baseline.invoke_json("put",&json!([key,key*2])).unwrap();}
        baseline.acknowledge_through(u64::MAX,u64::MAX);
        let bytes=baseline.checkpoint_portable().unwrap();
        for (action,args) in [("zero",json!([])),("get",json!([count])),("put",json!([count,77])),("erase",json!([count,true]))]{
            // Alternate order between sizes to reduce simple warmup/order bias.
            let (before,database_only,after)=if count==24{
                let a=measure(new.clone(),&bytes,Native,action,&args);
                let d=measure(new.clone(),&bytes,Unfused,action,&args);
                (measure(old.clone(),&bytes,Unfused,action,&args),d,a)
            }else{(measure(old.clone(),&bytes,Unfused,action,&args),measure(new.clone(),&bytes,Unfused,action,&args),measure(new.clone(),&bytes,Native,action,&args))};
            let runtime_only=measure(old.clone(),&bytes,Native,action,&args);
            // Correctness is checked separately from timing, including exact snapshots.
            let mut a=Session::restore(old.clone(),&bytes,Unfused).unwrap();
            let mut b=Session::restore(new.clone(),&bytes,Native).unwrap();
            for _ in 0..4 {let expected=baseline.invoke_json(action,&args).unwrap().json();assert_eq!(a.invoke_json(action,&args).unwrap(),expected);assert_eq!(b.invoke_json(action,&args).unwrap(),expected);assert_eq!(a.checkpoint().unwrap(),baseline.checkpoint_portable().unwrap());assert_eq!(b.checkpoint().unwrap(),a.checkpoint().unwrap());}
            baseline=Runtime::restore_portable(module.program().clone(),&bytes).unwrap();
            rows.push(json!({"rows":count,"action":action,"before":before,"database_only":database_only,"runtime_only":runtime_only,"after":after}));
        }
    }
    println!("{}",json!({"schema":1,"kind":"complete representation Session::invoke_json calls with acknowledgement; no checkpoint in timed region","source":module.identity().unwrap(),"old_representation":old.request().representation,"new_representation":new.request().representation,"generated_rust":if LIBRARY_PROFILE=="release"{"rustc -O -C lto=thin -C embed-bitcode=yes"}else{"rustc -O"},"library_profile":LIBRARY_PROFILE,"rounds":9,"calls_per_round":128,"measurements":rows}));
}
