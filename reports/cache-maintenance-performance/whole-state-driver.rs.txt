use std::{hint::black_box, time::Instant};
use verified_language::{core::CheckedModule, definition::CheckedDefinition, logic::{Context,Term}, representation::{self,CheckedRepresentation}, stateful::Runtime, syntax::parse};
use ink_runtime::representation::{Executor,Reference,Session};
use serde_json::{json,Value};
#[derive(Clone,Copy)]
struct Unfused;
impl Executor for Unfused {
    fn execute(&self,d:&CheckedDefinition,c:&Context,f:&str,a:Vec<Term>)->ink_core::LangResult<Term>{Native.execute(d,c,f,a)}
}
fn parity<E:Executor>(checked:CheckedRepresentation,module:&CheckedModule,executor:E){
    let mut session=Session::new(checked.clone(),executor).unwrap();
    let mut baseline=Runtime::new(module.program().clone()).unwrap();
    for i in 0..60u64 {
        let (action,args)=match i%10 {
            0=>("put",json!([0,i])),
            1=>("zero",json!([])),
            2=>("flag",json!([i%3,i%2==0])),
            3=>("flagged",json!([i%3])),
            4=>("erase",json!([0,true])),
            5=>("zero",json!([])),
            6=>("put",json!([i+1,i])),
            7=>("get",json!([i+1])),
            8=>("erase",json!([0,false])),
            _=>("zero",json!([])),
        };
        assert_eq!(session.invoke_json(action,&args).unwrap(),baseline.invoke_json(action,&args).unwrap().json());
        assert_eq!(session.checkpoint().unwrap(),baseline.checkpoint_portable().unwrap());
    }
}
fn measure<E:Executor>(checked:CheckedRepresentation,bytes:&[u8],executor:E,action:&str,args:&Value)->Value{
    let mut session=Session::restore(checked,bytes,executor).unwrap();
    for _ in 0..32 {black_box(session.invoke_json(action,args).unwrap());session.acknowledge_through(u64::MAX,u64::MAX);}
    let mut samples=vec![];
    for _ in 0..9 {
        let start=Instant::now();
        for _ in 0..128 {black_box(session.invoke_json(action,black_box(args)).unwrap());session.acknowledge_through(u64::MAX,u64::MAX);}
        samples.push(start.elapsed().as_nanos() as u64/128);
    }
    let mut sorted=samples.clone();sorted.sort();
    json!({"samples_ns":samples,"median_ns":sorted[4]})
}
fn main(){
    let module=CheckedModule::from_source(parse(SOURCE).unwrap()).unwrap();
    let load=|label:&str|{
        let p:Value=serde_json::from_slice(&std::fs::read(format!("{DIRECTORY}/{label}.json")).unwrap()).unwrap();
        representation::check(&module,&serde_json::from_value(p["requests"][1].clone()).unwrap()).unwrap()
    };
    let old=load("refresh");let new=load("preserve");
    parity(old.clone(),&module,Native);parity(new.clone(),&module,Native);
    parity(new.clone(),&module,Unfused);parity(new.clone(),&module,Reference);
    // Restores and transaction-boundary migration preserve the source snapshot.
    let mut base=Runtime::new(module.program().clone()).unwrap();
    base.invoke_json("put",&json!([0,99])).unwrap();
    let bytes=base.checkpoint_portable().unwrap();
    let mut migrated=Session::restore(old.clone(),&bytes,Native).unwrap();
    migrated.migrate(new.clone()).unwrap();assert_eq!(migrated.checkpoint().unwrap(),bytes);
    assert_eq!(migrated.invoke_json("zero",&json!([])).unwrap(),base.invoke_json("zero",&json!([])).unwrap().json());
    migrated.migrate(old.clone()).unwrap();assert_eq!(migrated.checkpoint().unwrap(),base.checkpoint_portable().unwrap());
    let layout=ink_core::snapshot::layout(module.program()).unwrap();
    let mut logical=ink_core::snapshot_wire::decode(&layout,&bytes,Default::default()).unwrap();logical.version=u64::MAX;
    let exhausted=ink_core::snapshot_wire::encode(&layout,&logical,Default::default()).unwrap();
    let mut session=Session::restore(new.clone(),&exhausted,Native).unwrap();
    let mut base=Runtime::restore_portable(module.program().clone(),&exhausted).unwrap();
    for (action,args) in [("put",json!([0,3])),("flag",json!([0,true])),("erase",json!([0,false]))]{
        assert!(base.invoke_json(action,&args).is_err());assert!(session.invoke_json(action,&args).is_err());
        assert_eq!(session.checkpoint().unwrap(),exhausted);
    }
    assert_eq!(session.invoke_json("erase",&json!([0,true])).unwrap(),base.invoke_json("erase",&json!([0,true])).unwrap().json());
    assert_eq!(session.checkpoint().unwrap(),exhausted);
    if !TIMED{return;}
    let mut rows=vec![];
    for count in [8,24,48]{
        let mut baseline=Runtime::new(module.program().clone()).unwrap();
        for key in 1..=count {baseline.invoke_json("put",&json!([key,key*2])).unwrap();}
        baseline.invoke_json("flag",&json!([0,true])).unwrap();
        baseline.acknowledge_through(u64::MAX,u64::MAX);
        let bytes=baseline.checkpoint_portable().unwrap();
        for (action,args) in [("erase",json!([count,true])),("flag",json!([0,false])),("put",json!([count,77])),("zero",json!([]))]{
            let mut a=Session::restore(old.clone(),&bytes,Native).unwrap();
            let mut b=Session::restore(new.clone(),&bytes,Native).unwrap();
            for _ in 0..4 {let expected=baseline.invoke_json(action,&args).unwrap().json();assert_eq!(a.invoke_json(action,&args).unwrap(),expected);assert_eq!(b.invoke_json(action,&args).unwrap(),expected);assert_eq!(a.checkpoint().unwrap(),baseline.checkpoint_portable().unwrap());assert_eq!(b.checkpoint().unwrap(),a.checkpoint().unwrap());}
            baseline=Runtime::restore_portable(module.program().clone(),&bytes).unwrap();
            let (before,after)=if count==24 {let after=measure(new.clone(),&bytes,Native,action,&args);(measure(old.clone(),&bytes,Native,action,&args),after)}else{(measure(old.clone(),&bytes,Native,action,&args),measure(new.clone(),&bytes,Native,action,&args))};
            let fallback_before=measure(old.clone(),&bytes,Unfused,action,&args);
            let fallback_after=measure(new.clone(),&bytes,Unfused,action,&args);
            rows.push(json!({"rows":count,"action":action,"before":before,"after":after,"fallback_before":fallback_before,"fallback_after":fallback_after}));
        }
    }
    println!("{}",json!({"schema":1,"kind":"DB-only cache maintenance; complete Session::invoke_json calls with acknowledgement","source":module.identity().unwrap(),"old_representation":old.request().representation,"new_representation":new.request().representation,"library_profile":LIBRARY_PROFILE,"rounds":9,"calls_per_round":128,"measurements":rows}));
}
