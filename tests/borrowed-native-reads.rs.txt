
#[cfg(test)]
mod borrowed_reads {
    use super::*;
    use std::alloc::{GlobalAlloc, Layout, System};
    use std::cell::Cell;
    thread_local! { static ALLOCATIONS: Cell<Option<usize>> = const { Cell::new(None) }; }
    struct Counting;
    unsafe impl GlobalAlloc for Counting {
        unsafe fn alloc(&self, layout: Layout) -> *mut u8 {
            ALLOCATIONS.with(|n| { if let Some(count) = n.get() { n.set(Some(count + 1)); } });
            System.alloc(layout)
        }
        unsafe fn dealloc(&self, ptr: *mut u8, layout: Layout) { System.dealloc(ptr, layout); }
        unsafe fn realloc(&self, ptr: *mut u8, layout: Layout, size: usize) -> *mut u8 {
            ALLOCATIONS.with(|n| { if let Some(count) = n.get() { n.set(Some(count + 1)); } });
            System.realloc(ptr, layout, size)
        }
    }
    #[global_allocator] static ALLOCATOR: Counting = Counting;
    fn call(ctx: &mut InkContext, op: &str, args: &[u64]) -> u64 {
        let op = std::ffi::CString::new(op).unwrap();
        unsafe { ink_op(ctx, op.as_ptr(), args.as_ptr(), args.len()) }
    }
    fn without_allocations(ctx: &mut InkContext, op: &'static [u8], args: &[u64]) -> u64 {
        ALLOCATIONS.with(|n| n.set(Some(0)));
        let result = unsafe { ink_op(ctx, op.as_ptr().cast(), args.as_ptr(), args.len()) };
        let allocations = ALLOCATIONS.with(|n| n.replace(None).unwrap());
        assert_eq!(allocations, 0, "readonly operation copied/allocated its operands");
        result
    }
    fn large() -> V { V::List((0..8192).map(|i| V::Record("Row".into(), BTreeMap::from([
        ("number".into(), V::U32(i)), ("text".into(), V::String("payload".repeat(16))),
    ]))).collect()) }
    #[test]
    fn aggregate_inspection_and_comparison_do_not_allocate() {
        let mut state = State::new(); let c = &mut state.ctx;
        let list = c.put(large());
        let other = c.put(large());
        let record = c.put(V::Record("Envelope".into(), BTreeMap::from([
            ("payload".into(), large()), ("count".into(), V::U32(8192)),
        ])));
        let sum = c.put(V::Option(Some(Box::new(large()))));
        let fuel = c.fuel; let arena = c.arena_bytes; let slots = c.values.len();
        let h = without_allocations(c, b"list_length\0", &[list]);
        assert_eq!(c.value(h).unwrap(), V::U32(8192));
        let h = without_allocations(c, b"field:count\0", &[record]);
        assert_eq!(c.value(h).unwrap(), V::U32(8192));
        let h = without_allocations(c, b"tag:Some\0", &[sum]);
        assert_eq!(c.value(h).unwrap(), V::Bool(true));
        let h = without_allocations(c, b"==\0", &[list, other]);
        assert_eq!(c.value(h).unwrap(), V::Bool(true));
        let h = without_allocations(c, b"<=\0", &[list, other]);
        assert_eq!(c.value(h).unwrap(), V::Bool(true));
        assert_eq!(c.fuel, fuel - 5);
        assert_eq!((c.arena_bytes, c.values.len()), (arena, slots));
    }
    #[test]
    fn selected_composite_is_owned_and_input_remains_unchanged() {
        let mut state = State::new(); let c = &mut state.ctx;
        let input = V::Record("Envelope".into(), BTreeMap::from([
            ("payload".into(), V::List(vec![V::String("before".into())])),
        ]));
        let h = c.put(input.clone()); let selected = call(c, "field:payload", &[h]);
        let V::List(xs) = &mut c.values[selected as usize - 1] else { panic!() };
        xs.push(V::String("after".into()));
        assert_eq!(c.value(h).unwrap(), input);
    }
    #[test]
    fn comparisons_preserve_ieee_scalars_and_bitwise_aggregates() {
        let mut state = State::new(); let c = &mut state.ctx;
        for (a, b, eq) in [(0, 0x80000000, true), (0x7fc12345, 0x7fc12345, false),
                          (0x7f800000, 0x7f800000, true), (1, 0, false)] {
            for (left, right, expected) in [
                (V::F32(a), V::F32(b), eq),
                (V::List(vec![V::F32(a)]), V::List(vec![V::F32(b)]), a == b),
            ] {
                let x = c.put(left.clone()); let y = c.put(right.clone());
                for op in ["==", "!=", "<", ">", "<=", ">="] {
                    let original = c.primitive(op, &mut [left.clone(), right.clone()]).unwrap();
                    let result = call(c, op, &[x, y]);
                    assert_eq!(c.value(result).unwrap(), original);
                    if op == "==" { assert_eq!(original, V::Bool(expected)); }
                }
            }
        }
    }
    #[test]
    fn invalid_handles_errors_and_fuel_stay_checked() {
        for (op, input, error) in [
            ("list_length", V::U32(1), "list length"),
            ("field:x", V::Unit, "field requires record/vector"),
            ("field:z", V::Vector(vec![V::U32(1)]), "unknown component"),
            ("field:missing", V::Record("Empty".into(), BTreeMap::new()), "unknown field"),
        ] {
            let mut state = State::new(); let h = state.ctx.put(input);
            assert_eq!(call(&mut state.ctx, op, &[h]), 0);
            assert!(matches!(&state.ctx.error, Some(Failure::Host(e)) if e == error));
        }
        for invalid in [0, 123, 6u64 << 32, (2u64 << 32) | 2] {
            let mut state = State::new(); let c = &mut state.ctx;
            assert_eq!(call(c, "list_length", &[invalid]), 0);
            assert!(matches!(&c.error, Some(Failure::Host(e)) if e == "invalid value handle"));
        }
        let mut state = State::new(); let c = &mut state.ctx;
        let h = c.put(V::List(vec![])); c.fuel = 0;
        assert_eq!(call(c, "list_length", &[h]), 0);
        assert!(matches!(&c.error, Some(Failure::Host(e)) if e == "Ink execution budget exhausted"));
        assert_eq!(call(c, "==", &[h, h]), 0); // first failure remains sticky
        assert_eq!(c.fuel, 0);
    }
}
